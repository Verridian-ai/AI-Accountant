- [ ] Add `budgetsApi` object:
```typescript
export const budgetsApi = {
  create: (params: any) =>
    fetch(`${API_URL}/budgets`, { method: 'POST', headers: { ...getAuthHeaders(), 'Content-Type': 'application/json' }, body: JSON.stringify(params) }).then(r => r.json()),
  list: (status?: string) =>
    fetch(`${API_URL}/budgets?userId=default${status ? `&status=${status}` : ''}`, { headers: getAuthHeaders() }).then(r => r.json()),
  get: (id: string) =>
    fetch(`${API_URL}/budgets/${id}`, { headers: getAuthHeaders() }).then(r => r.json()),
  update: (id: string, updates: any) =>
    fetch(`${API_URL}/budgets/${id}`, { method: 'PUT', headers: { ...getAuthHeaders(), 'Content-Type': 'application/json' }, body: JSON.stringify(updates) }).then(r => r.json()),
  delete: (id: string) =>
    fetch(`${API_URL}/budgets/${id}`, { method: 'DELETE', headers: getAuthHeaders() }).then(r => r.json()),
  addLine: (id: string, line: any) =>
    fetch(`${API_URL}/budgets/${id}/lines`, { method: 'POST', headers: { ...getAuthHeaders(), 'Content-Type': 'application/json' }, body: JSON.stringify(line) }).then(r => r.json()),
  getVariance: (id: string) =>
    fetch(`${API_URL}/budgets/${id}/variance`, { headers: getAuthHeaders() }).then(r => r.json()),
  getVarianceSummary: (id: string) =>
    fetch(`${API_URL}/budgets/${id}/variance/summary`, { headers: getAuthHeaders() }).then(r => r.json()),
};
```
