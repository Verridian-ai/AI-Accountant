```python
class TransactionNode(DataPoint):

    amount_cents: int
    category: str
    date: str
    gst_applicable: bool
    claim_type: str

class MerchantNode(DataPoint):

    canonical_name: str
    abn: str
    category_pattern: list["CategoryNode"]
    metadata: dict = {"index_fields": ["name", "canonical_name"]}
```
